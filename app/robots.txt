import { MetadataRoute } from 'next'

export default function robots(): MetadataRoute.Robots {
  const baseUrl = process.env.NEXT_PUBLIC_APP_URL || 'https://truth-app-delta.vercel.app'

  return {
    rules: [
      {
        userAgent: '*',
        allow: '/',
        disallow: [
          '/api/',
          '/admin/',
          '/settings/',
          '/inbox/',
          '/* /edit',
          '/* /delete',
        ],
      },
    ],
    sitemap: `${baseUrl}/sitemap.xml`,
  }
}
